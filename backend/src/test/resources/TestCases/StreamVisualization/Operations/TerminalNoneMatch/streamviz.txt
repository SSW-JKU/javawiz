stream 1
  lanes
    of | int |
    map | int | (n)->n * 2
    noneMatch | boolean | (n)->n > 10
  playback
    step 1: of@1 START int "1"
    step 2: map@1 OUT int "2"
    step 3: noneMatch@1 END boolean "true"
    step 4: of@2 START int "2"
    step 5: map@2 OUT int "4"
    step 6: noneMatch@2 END boolean "true"
    step 7: of@3 START int "3"
    step 8: map@3 OUT int "6"
    step 9: noneMatch@3 END boolean "true"
    link from step 2: of@1 -> map@1
    link from step 3: map@1 -> noneMatch@1
    link from step 5: of@2 -> map@2
    link from step 6: map@1 -> noneMatch@2
    link from step 6: map@2 -> noneMatch@2
    link from step 8: of@3 -> map@3
    link from step 9: map@1 -> noneMatch@3
    link from step 9: map@2 -> noneMatch@3
    link from step 9: map@3 -> noneMatch@3
  final
    of@1 START int "1"
    of@2 START int "2"
    of@3 START int "3"
    map@1 OUT int "2"
    map@2 OUT int "4"
    map@3 OUT int "6"
    noneMatch@3 END boolean "true"
    link of@1 -> map@1
    link map@1 -> noneMatch@3
    link of@2 -> map@2
    link map@2 -> noneMatch@3
    link of@3 -> map@3
    link map@3 -> noneMatch@3
